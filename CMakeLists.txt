cmake_minimum_required(VERSION 3.22)
project(CueFlow VERSION 0.5.4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE-8.0.12" CACHE PATH "JUCE source")
add_subdirectory(${JUCE_DIR} juce EXCLUDE_FROM_ALL)
add_library(CueCore STATIC src/Lyrics.cpp src/State.cpp src/QrcDecode.cpp src/Alignment.cpp src/vendor/qrcodegen/qrcodegen.cpp)
target_include_directories(CueCore PUBLIC src)
target_compile_definitions(CueCore PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(CueCore PUBLIC juce::juce_core)
juce_add_console_app(CueFlowTests PRODUCT_NAME "CueFlow tests")
target_sources(CueFlowTests PRIVATE tests/CoreTests.cpp)
target_link_libraries(CueFlowTests PRIVATE CueCore)
enable_testing()
add_test(NAME core COMMAND CueFlowTests tests/fixtures)
set(CUEFLOW_FORMATS VST3)
if(APPLE)
 list(APPEND CUEFLOW_FORMATS AU)
endif()
option(CUEFLOW_BUILD_AAX "Build unsigned AAX development bundle (not a signed Pro Tools release)" OFF)
if(CUEFLOW_BUILD_AAX)
 list(APPEND CUEFLOW_FORMATS AAX)
endif()
option(CUEFLOW_BUILD_CLAP "Build CLAP plugin" ON)
juce_add_plugin(CueFlowPlugin COMPANY_NAME "CueFlow" BUNDLE_ID "studio.cueflow.prompter"
 PLUGIN_MANUFACTURER_CODE Cufl PLUGIN_CODE Cfp1 FORMATS ${CUEFLOW_FORMATS}
 PRODUCT_NAME "CueFlow" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
 COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Tools)
target_sources(CueFlowPlugin PRIVATE src/Processor.cpp src/Editor.cpp src/Reader.cpp src/Projection.cpp src/AlignmentPanel.cpp)
target_compile_definitions(CueFlowPlugin PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(CueFlowPlugin PUBLIC CueCore juce::juce_audio_utils)
if(CUEFLOW_BUILD_CLAP)
 if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/clap-juce-extensions/clap-libs/clap/include/clap/clap.h")
  message(FATAL_ERROR "CLAP sources missing. Run scripts/setup-formats.ps1 or scripts/setup-formats.sh; alternatively set CUEFLOW_BUILD_CLAP=OFF.")
 endif()
 add_subdirectory(third_party/clap-juce-extensions EXCLUDE_FROM_ALL)
 clap_juce_extensions_plugin(TARGET CueFlowPlugin CLAP_ID "studio.cueflow.prompter" CLAP_FEATURES "audio-effect" "utility")
 add_subdirectory(tests/clap-smoke)
 if(WIN32)
  add_test(NAME clap_smoke COMMAND CueFlowClapSmoke "CueFlowPlugin_artefacts/$<CONFIG>/CLAP/CueFlow.clap")
  set_tests_properties(clap_smoke PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
 elseif(APPLE)
  add_test(NAME clap_smoke COMMAND CueFlowClapSmoke $<TARGET_FILE:CueFlowPlugin_CLAP>)
 endif()
endif()
juce_add_console_app(CueFlowIntegration PRODUCT_NAME "CueFlowIntegration")
target_sources(CueFlowIntegration PRIVATE tests/ProcessorTests.cpp)
target_link_libraries(CueFlowIntegration PRIVATE CueFlowPlugin)
add_test(NAME integration COMMAND CueFlowIntegration)
juce_add_gui_app(CueFlowStandalone PRODUCT_NAME "CueFlow" COMPANY_NAME "CueFlow" BUNDLE_ID "studio.cueflow.standalone")
target_sources(CueFlowStandalone PRIVATE src/Standalone.cpp)
target_link_libraries(CueFlowStandalone PRIVATE CueFlowPlugin)
set_tests_properties(core integration PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")

juce_add_console_app(CueFlowKaraokeTests PRODUCT_NAME "CueFlow karaoke tests")
target_sources(CueFlowKaraokeTests PRIVATE tests/KaraokeTests.cpp)
target_link_libraries(CueFlowKaraokeTests PRIVATE CueCore juce::juce_gui_basics)
add_test(NAME karaoke COMMAND CueFlowKaraokeTests)
